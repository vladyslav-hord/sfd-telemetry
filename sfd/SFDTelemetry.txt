public void OnStartup()
{
    if (string.IsNullOrEmpty(s_serverSessionId))
    {
        s_serverSessionId = Guid.NewGuid().ToString("D");
        s_sequence = 0;
        s_sessions.Clear();
        s_playerSessions.Clear();
    }
    m_roundId = Guid.NewGuid().ToString("D");
    m_roundStartMs = Game.TotalElapsedGameTime;
    m_lastGameOver = Game.IsGameOver;
    m_lastSuddenDeath = Game.SuddenDeathActive;
    m_sceneStates.Clear();
    m_sceneCaptureUntil.Clear();
    m_sceneRing.Clear();
    m_pendingSceneCaptures.Clear();
    m_sceneTransportTokens = SCENE_TRANSPORT_BURST_BYTES;
    m_sceneTransportLastMs = Game.TotalElapsedGameTime;
    m_lastSceneHealthMs = Game.TotalElapsedGameTime;
    m_transport = Game.GetSharedStorage(STORAGE_NAME);

    Game.Events.StartUserJoinCallback(OnUserJoin);
    Game.Events.StartUserLeaveCallback(OnUserLeave);
    Game.Events.StartUserMessageCallback(OnUserMessage);
    Game.Events.StartPlayerCreatedCallback(OnPlayerCreated);
    Game.Events.StartPlayerDamageCallback(OnPlayerDamage);
    Game.Events.StartPlayerDeathCallback(OnPlayerDeath);
    Game.Events.StartPlayerMeleeActionCallback(OnPlayerMeleeAction);
    Game.Events.StartPlayerWeaponAddedActionCallback(OnWeaponAdded);
    Game.Events.StartPlayerWeaponRemovedActionCallback(OnWeaponRemoved);
    if (ENABLE_INPUTS) Game.Events.StartPlayerKeyInputCallback(OnPlayerKeyInput);
    if (ENABLE_PROJECTILES)
    {
        Game.Events.StartProjectileCreatedCallback(OnProjectileCreated);
        Game.Events.StartProjectileHitCallback(OnProjectileHit);
        Game.Events.StartExplosionHitCallback(OnExplosionHit);
    }
    if (ENABLE_OBJECT_EVENTS)
    {
        Game.Events.StartObjectCreatedCallback(OnObjectCreated);
        Game.Events.StartObjectDamageCallback(OnObjectDamage);
        Game.Events.StartObjectTerminatedCallback(OnObjectTerminated);
    }
    Game.Events.StartUpdateCallback(OnTelemetryUpdate, HIGHRES_STATE_INTERVAL_MS);

    QueueEvent("script_start", "", ObjectJson(
        Field("script_version", Quote(SCRIPT_VERSION)),
        Field("sfd_build", Quote(SFD_BUILD)),
        Field("transport", Quote("shared_storage_base64_spool")),
        Field("map_name", Quote(Game.MapName)),
        Field("config", ObjectJson(
            Field("baseline_state_hz", Number(1000f / BASELINE_STATE_INTERVAL_MS)),
            Field("highres_state_hz", Number(1000f / HIGHRES_STATE_INTERVAL_MS)),
            Field("highres_window_seconds", Number(HIGHRES_WINDOW_MS / 1000f)),
            Field("ping_interval_seconds", Number(PING_INTERVAL_MS / 1000f)),
            Field("enable_chat", Bool(ENABLE_CHAT)),
            Field("enable_inputs", Bool(ENABLE_INPUTS)),
            Field("enable_projectiles", Bool(ENABLE_PROJECTILES)),
            Field("enable_object_events", Bool(ENABLE_OBJECT_EVENTS)),
            Field("enable_world_context", Bool(ENABLE_WORLD_CONTEXT)),
            Field("verbose_debug", Bool(VERBOSE_DEBUG))))));

    IUser[] users = Game.GetActiveUsers();
    for (int i = 0; i < users.Length; i++) EnsureSession(users[i], "user_present");
    QueueRoundStart();
    QueueSceneManifest();
    SnapshotAllStatistics("round_start");
    ForceFlush();
}

public void OnShutdown()
{
    FlushObjectDamageBatches();
    SnapshotAllStatistics("round_end");
    EndRound("script_shutdown");
    QueueHealth();
    QueueEvent("script_shutdown", "", ObjectJson(
        Field("map_name", Quote(Game.MapName)),
        Field("map_round", Game.MapRound.ToString()),
        Field("is_game_over", Bool(Game.IsGameOver))));
    ForceFlush();
}

private void OnTelemetryUpdate(float elapsed)
{
    float now = Game.TotalElapsedGameTime;
    if (now - m_lastHighresMs >= HIGHRES_STATE_INTERVAL_MS)
    {
        m_lastHighresMs = now;
        SampleStates(now);
        SampleSceneRing(now);
        FlushSceneCaptures(now);
    }
    if (now - m_lastBaselineMs >= BASELINE_STATE_INTERVAL_MS)
    {
        m_lastBaselineMs = now;
        PersistBaselineStates();
    }
    if (now - m_lastPingMs >= PING_INTERVAL_MS)
    {
        m_lastPingMs = now;
        SamplePing();
    }
    if (ENABLE_SCENE_TELEMETRY && now - m_lastSceneMs >= SCENE_DYNAMIC_INTERVAL_MS)
    {
        m_lastSceneMs = now;
        SampleSceneDynamics(now);
    }
    if (now - m_lastSceneHealthMs >= 60000f)
    {
        m_lastSceneHealthMs = now;
        QueueHealth();
    }
    if (Game.IsGameOver && !m_lastGameOver) EndRound("game_over_transition");
    m_lastGameOver = Game.IsGameOver;
    if (Game.SuddenDeathActive != m_lastSuddenDeath)
    {
        m_lastSuddenDeath = Game.SuddenDeathActive;
        QueueEvent("sudden_death_transition", "", ObjectJson(
            Field("active", Bool(m_lastSuddenDeath)),
            Field("enabled", Bool(Game.SuddenDeathEnabled))));
    }
    if (now - m_lastFlushMs >= STORAGE_FLUSH_INTERVAL_MS)
    {
        m_lastFlushMs = now;
        FlushObjectDamageBatches();
        FlushPending();
    }
}

private void OnUserJoin(IUser[] users)
{
    for (int i = 0; i < users.Length; i++) EnsureSession(users[i], "user_join");
}

private void OnUserLeave(IUser[] users, DisconnectionType type)
{
    for (int i = 0; i < users.Length; i++)
    {
        IUser user = users[i];
        PlayerSession session = GetSession(user);
        if (session == null) session = CreateSession(user);
        IPlayer player = user.GetPlayer();
        if (player != null) SnapshotStatistics(player, "leave");
        double durationMs = (DateTime.UtcNow - session.JoinedUtc).TotalMilliseconds;
        QueueEvent("user_leave", session.Id, ObjectJson(
            Field("reason", Quote(type.ToString())),
            Field("duration_ms", Number(durationMs)),
            Field("ping", user.Ping.ToString()),
            Field("spectating", Bool(user.Spectating)),
            Field("team", Quote(user.GetTeam().ToString())),
            Field("total_games", user.TotalGames.ToString()),
            Field("total_wins", user.TotalWins.ToString()),
            Field("total_losses", user.TotalLosses.ToString()),
            Field("state", player == null ? "null" : PlayerStateJson(player, 0))));
        s_sessions.Remove(user.UserIdentifier);
        if (player != null) s_playerSessions.Remove(player.UniqueID);
        m_ring.Remove(session.Id);
        m_captureUntil.Remove(session.Id);
    }
}

private void OnUserMessage(UserMessageCallbackArgs args)
{
    if (!ENABLE_CHAT || args == null || args.User == null) return;
    PlayerSession session = GetOrCreateSession(args.User);
    IPlayer player = args.User.GetPlayer();
    QueueEvent("chat_message", session.Id, ObjectJson(
        Field("account_name", Quote(args.User.AccountName)),
        Field("character_name", Quote(args.User.Name)),
        Field("message", Quote(args.Message)),
        Field("is_command", Bool(args.IsCommand)),
        Field("command", Quote(args.Command)),
        Field("command_arguments", Quote(args.CommandArguments)),
        Field("map_name", Quote(Game.MapName)),
        Field("player_count", Game.GetPlayers().Length.ToString()),
        Field("team", Quote(args.User.GetTeam().ToString())),
        Field("spectating", Bool(args.User.Spectating)),
        Field("state", player == null ? "null" : PlayerStateJson(player, 0))));
}

private void OnPlayerCreated(IPlayer[] players)
{
    for (int i = 0; i < players.Length; i++)
    {
        IPlayer player = players[i];
        string sessionId = SessionId(player);
        QueueEvent("player_created", sessionId, PlayerContextJson(player));
        SnapshotStatistics(player, "player_created");
    }
}

private void OnPlayerDamage(IPlayer victim, PlayerDamageArgs args)
{
    IPlayer attacker = ResolveDamageAttacker(args);
    string victimSession = SessionId(victim);
    string attackerSession = SessionId(attacker);
    float distance = Distance(attacker, victim);
    QueueEvent("player_damage", victimSession, ObjectJson(
        Field("victim_session_id", Quote(victimSession)),
        Field("attacker_session_id", Quote(attackerSession)),
        Field("damage", Number(args.Damage)),
        Field("damage_type", Quote(args.DamageType.ToString())),
        Field("source_id", args.SourceID.ToString()),
        Field("overkill", Bool(args.OverkillDamage)),
        Field("distance", distance < 0 ? "null" : Number(distance)),
        Field("victim", PlayerContextJson(victim)),
        Field("attacker", attacker == null ? "null" : PlayerContextJson(attacker))));
    if (args.Damage >= 20f) TriggerHighres(victimSession, "large_damage");
}

private void OnPlayerDeath(IPlayer player, PlayerDeathArgs args)
{
    string sessionId = SessionId(player);
    QueueEvent("player_death", sessionId, ObjectJson(
        Field("victim_session_id", Quote(sessionId)),
        Field("event", Quote(args.Event.ToString())),
        Field("killed", Bool(args.Killed)),
        Field("removed", Bool(args.Removed)),
        Field("state", PlayerContextJson(player))));
    SnapshotStatistics(player, "death");
    TriggerHighres(sessionId, "death");
}

private void OnPlayerMeleeAction(IPlayer attacker, PlayerMeleeHitArg[] args)
{
    string attackerSession = SessionId(attacker);
    int focusObjectId = 0;
    StringBuilder hits = new StringBuilder("[");
    for (int i = 0; i < args.Length; i++)
    {
        if (i > 0) hits.Append(',');
        IPlayer victim = args[i].HitObject as IPlayer;
        if (!args[i].IsPlayer)
        {
            MarkSceneObject(args[i].ObjectID);
            if (focusObjectId == 0) focusObjectId = args[i].ObjectID;
        }
        hits.Append(ObjectJson(
            Field("object_id", args[i].ObjectID.ToString()),
            Field("is_player", Bool(args[i].IsPlayer)),
            Field("victim_session_id", Quote(SessionId(victim))),
            Field("damage", Number(args[i].HitDamage)),
            Field("x", Number(args[i].HitPosition.X)),
            Field("y", Number(args[i].HitPosition.Y))));
        if (victim != null) TriggerHighres(SessionId(victim), "melee_hit");
    }
    hits.Append(']');
    QueueEvent("melee_action", attackerSession, ObjectJson(
        Field("attacker_session_id", Quote(attackerSession)),
        Field("action", Quote(attacker.IsKicking ? "kick" : "melee")),
        Field("hit_count", args.Length.ToString()),
        Field("hits", hits.ToString()),
        Field("attacker", PlayerContextJson(attacker))));
    if (args.Length > 0) TriggerHighres(attackerSession, "melee_hit");
    if (args.Length > 0) QueueSceneWindow(attacker.IsKicking ? "kick" : "melee_hit", focusObjectId);
}

private void OnPlayerKeyInput(IPlayer player, VirtualKeyInfo[] keys)
{
    string sessionId = SessionId(player);
    if (keys.Length == 0) return;
    StringBuilder transitions = new StringBuilder("[");
    for (int i = 0; i < keys.Length; i++)
    {
        if (i > 0) transitions.Append(',');
        transitions.Append(ObjectJson(
            Field("key", Quote(keys[i].Key.ToString())),
            Field("event", Quote(keys[i].Event.ToString())),
            Field("value", keys[i].Value.ToString())));
        if (keys[i].Event == VirtualKeyEvent.Pressed && keys[i].Key == VirtualKey.GRAB)
            TriggerHighres(sessionId, "grab_input");
        if (keys[i].Event == VirtualKeyEvent.Pressed && keys[i].Key == VirtualKey.ATTACK &&
            (player.IsGrabbing || player.IsHoldingPlayerInGrab || player.IsThrowing))
            TriggerHighres(sessionId, "throw_input");
    }
    transitions.Append(']');
    Vector2 pos = player.GetWorldPosition();
    QueueEvent("key_input_batch", sessionId, ObjectJson(
        Field("transitions", transitions.ToString()),
        Field("team", Quote(player.GetTeam().ToString())),
        Field("x", Number(pos.X)), Field("y", Number(pos.Y))));
}

    {

private void OnObjectDamage(IObject obj, ObjectDamageArgs args)
{
    MarkSceneObject(obj.UniqueID);
    IPlayer sourcePlayer = args.IsPlayer ? Game.GetPlayer(args.SourceID) : null;
    m_pendingObjectDamage.Add(ObjectDamageJson(obj,
        Field("damage", Number(args.Damage)), Field("damage_type", Quote(args.DamageType.ToString())),
        Field("source_id", args.SourceID.ToString()), Field("source_is_player", Bool(args.IsPlayer)),
        Field("source_player_session_id", Quote(SessionId(sourcePlayer))),
        Field("health_after", Number(obj.GetHealth())), Field("game_ms", Number(Game.TotalElapsedGameTime))));
    QueueSceneWindow("object_damage", obj.UniqueID);
}

private void FlushObjectDamageBatches()
{
    while (m_pendingObjectDamage.Count > 0)
    {
        int count = Math.Min(OBJECT_DAMAGE_BATCH_SIZE, m_pendingObjectDamage.Count);
        StringBuilder records = new StringBuilder("[");
        for (int i = 0; i < count; i++)
        {
            if (i > 0) records.Append(',');
            records.Append(m_pendingObjectDamage[i]);
        }
        records.Append(']');
        m_pendingObjectDamage.RemoveRange(0, count);
        QueueEvent("object_damage_batch", "", ObjectJson(Field("records", records.ToString())));
    }
}

private string ObjectDamageJson(IObject obj, params string[] fields)
{
    Vector2 pos = obj.GetWorldPosition();
    Vector2 velocity = obj.GetLinearVelocity();
    List<string> values = new List<string>();
    values.Add(Field("object_id", obj.UniqueID.ToString()));
    values.Add(Field("x", Number(pos.X))); values.Add(Field("y", Number(pos.Y)));
    values.Add(Field("vx", Number(velocity.X))); values.Add(Field("vy", Number(velocity.Y)));
    values.Add(Field("health", Number(obj.GetHealth()))); values.Add(Field("max_health", Number(obj.GetMaxHealth())));
    values.Add(Field("is_missile", Bool(obj.IsMissile)));
    for (int i = 0; i < fields.Length; i++) values.Add(fields[i]);
    return ObjectJson(values.ToArray());
}

private void OnObjectTerminated(IObject[] objects)

        QueueEvent("object_terminated", "", ObjectJson(Field("object", SceneObjectJson(objects[i]))));

private void QueueSceneManifest()
{
    if (!ENABLE_SCENE_TELEMETRY) return;
    IObject[] objects = Game.GetObjects<IObject>();
    int limit = Math.Min(objects.Length, MAX_TRACKED_OBJECTS);
    for (int offset = 0; offset < limit; offset += 25)
    {
        StringBuilder batch = new StringBuilder("[");
        int end = Math.Min(offset + 25, limit);
        for (int i = offset; i < end; i++)
        {
            if (i > offset) batch.Append(',');
            batch.Append(SceneObjectJson(objects[i]));
        }
        batch.Append(']');
        QueueEvent("scene_manifest_batch", "", ObjectJson(
            Field("batch_offset", offset.ToString()), Field("object_total", limit.ToString()),
            Field("entities", batch.ToString())));
    }
}

private void SampleSceneDynamics(float now)
{
    IObject[] objects = Game.GetObjects<IObject>();
    IPlayer[] players = Game.GetPlayers();
    StringBuilder changed = new StringBuilder("[");
    int added = 0;
    for (int pass = 0; pass < 2 && added < MAX_SCENE_FRAME_OBJECTS; pass++)
    for (int i = 0; i < objects.Length && i < MAX_TRACKED_OBJECTS && added < MAX_SCENE_FRAME_OBJECTS; i++)
    {
        IObject obj = objects[i];
        if (obj == null || obj.IsRemoved) continue;
        Vector2 pos = obj.GetWorldPosition();
        Vector2 velocity = obj.GetLinearVelocity();
        float angle = obj.GetAngle();
        float angular = obj.GetAngularVelocity();
        float health = obj.GetHealth();
        SceneState old;
        bool known = m_sceneStates.TryGetValue(obj.UniqueID, out old);
        bool moving = velocity.Length() >= 0.5f || Math.Abs(angular) >= 0.1f || obj.IsMissile;
        if ((pass == 0) != IsScenePriority(obj, moving, now, players)) continue;
        float heartbeat = moving ? SCENE_DYNAMIC_HEARTBEAT_MS : SCENE_STATIONARY_HEARTBEAT_MS;
        bool changedState = !known || (pos - old.Position).Length() >= 0.25f ||
            (velocity - old.Velocity).Length() >= 0.5f || Math.Abs(angle - old.Angle) >= 0.05f ||
            Math.Abs(angular - old.AngularVelocity) >= 0.1f || Math.Abs(health - old.Health) >= 0.01f ||
            obj.IsMissile != old.Missile || now - old.LastSavedMs >= heartbeat;
        if (!changedState) continue;
        if (added >= MAX_SCENE_FRAME_OBJECTS)
        {
            m_sceneDroppedSampleCount++;
            continue;
        }
        if (added > 0) changed.Append(',');
        changed.Append(SceneFrameJson(obj));
        m_sceneStates[obj.UniqueID] = new SceneState { Position = pos, Velocity = velocity, Angle = angle,
            AngularVelocity = angular, Health = health, Missile = obj.IsMissile, LastSavedMs = now };
        added++;
    }
    if (added > 0)
    {
        changed.Append(']');
        QueueEvent("scene_frame_batch", "", ObjectJson(Field("entities", changed.ToString())));
    }
    // Projectile callbacks are exact contacts; these short 4 Hz trajectories supply
    // the motion context between create and hit without polling the whole world at 20 Hz.
    IProjectile[] projectiles = Game.GetProjectiles();
    StringBuilder projectileFrames = new StringBuilder("[");
    int projectileCount = Math.Min(projectiles.Length, MAX_PROJECTILE_FRAME_SAMPLES);
    for (int i = 0; i < projectileCount; i++)
    {
        if (i > 0) projectileFrames.Append(',');
        projectileFrames.Append(ProjectileJson(projectiles[i]));
    }
    projectileFrames.Append(']');
    if (projectileCount > 0)
        QueueEvent("scene_frame_batch", "", ObjectJson(Field("entities", projectileFrames.ToString())));
}

private void SampleSceneRing(float now)
{
    if (!ENABLE_SCENE_TELEMETRY) return;
    IObject[] objects = Game.GetObjects<IObject>();
    IPlayer[] players = Game.GetPlayers();
    int tracked = 0;
    for (int pass = 0; pass < 2 && tracked < MAX_HIGHRES_OBJECTS; pass++)
    for (int i = 0; i < objects.Length && i < MAX_TRACKED_OBJECTS && tracked < MAX_HIGHRES_OBJECTS; i++)
    {
        IObject obj = objects[i];
        if (obj == null || obj.IsRemoved) continue;
        bool moving = obj.GetLinearVelocity().Length() >= 0.5f || Math.Abs(obj.GetAngularVelocity()) >= 0.1f || obj.IsMissile;
        if ((pass == 0) != IsScenePriority(obj, moving, now, players)) continue;
        List<ScenePoint> points;
        if (!m_sceneRing.TryGetValue(obj.UniqueID, out points))
        {
            points = new List<ScenePoint>();
            m_sceneRing[obj.UniqueID] = points;
        }
        points.Add(new ScenePoint { GameMs = now, Json = SceneFrameJson(obj) });
        while (points.Count > SCENE_HIGHRES_RING_SAMPLES) points.RemoveAt(0);
        tracked++;
    }
}

private void MarkSceneObject(int objectId)
{
    if (objectId > 0) m_sceneCaptureUntil[objectId] = Game.TotalElapsedGameTime + HIGHRES_POST_EVENT_MS;
}

private bool IsScenePriority(IObject obj, bool moving, float now)
{
    return IsScenePriority(obj, moving, now, Game.GetPlayers());
}

private bool IsScenePriority(IObject obj, bool moving, float now, IPlayer[] players)
{
    float captureUntil;
    if (m_sceneCaptureUntil.TryGetValue(obj.UniqueID, out captureUntil) && captureUntil >= now) return true;
    if (obj.IsMissile) return true;
    Vector2 position = obj.GetWorldPosition();
    for (int i = 0; i < players.Length; i++)
        if (players[i] != null && (players[i].GetWorldPosition() - position).Length() <= SCENE_NEAR_PLAYER_RADIUS) return true;
    return moving;
}

private void QueueSceneWindow(string trigger)
{
    QueueSceneWindow(trigger, 0);
}

private void QueueSceneWindow(string trigger, int focusObjectId)
{
    if (!ENABLE_SCENE_TELEMETRY) return;
    float now = Game.TotalElapsedGameTime;
    if (now - m_lastSceneWindowMs < SCENE_WINDOW_COOLDOWN_MS)
    {
        m_sceneWindowSuppressedCount++;
        return;
    }
    m_lastSceneWindowMs = now;
    IObject[] objects = Game.GetObjects<IObject>();
    StringBuilder entities = new StringBuilder("[");
    int count = Math.Min(objects.Length, MAX_SCENE_WINDOW_OBJECTS);
    for (int i = 0; i < count; i++)
    {
        if (i > 0) entities.Append(',');
        entities.Append(SceneFrameJson(objects[i]));
    }
    entities.Append(']');
    QueueEvent("scene_window_complete", "", ObjectJson(
        Field("trigger", Quote(trigger)), Field("coverage", "1"), Field("entities", entities.ToString())));
    if (focusObjectId > 0 && now - m_lastSceneHighresCaptureMs >= SCENE_HIGHRES_CAPTURE_COOLDOWN_MS && m_pending.Count < MAX_PENDING_EVENTS * 3 / 4)
    {
        m_lastSceneHighresCaptureMs = now;
        m_pendingSceneCaptures.Add(new SceneWindowCapture { ObjectId = focusObjectId, DumpAtMs = now + HIGHRES_POST_EVENT_MS, Trigger = trigger });
    }
    else if (focusObjectId > 0)
    {
        m_sceneWindowSuppressedCount++;
    }
}

private void FlushSceneCaptures(float now)
{
    for (int i = m_pendingSceneCaptures.Count - 1; i >= 0; i--)
    {
        SceneWindowCapture capture = m_pendingSceneCaptures[i];
        if (now < capture.DumpAtMs) continue;
        DumpSceneRing(capture.ObjectId, capture.Trigger);
        m_pendingSceneCaptures.RemoveAt(i);
    }
}

private void DumpSceneRing(int objectId, string trigger)
{
    List<ScenePoint> points;
    if (!m_sceneRing.TryGetValue(objectId, out points) || points.Count == 0) return;
    int first = Math.Max(0, points.Count - SCENE_HIGHRES_RING_SAMPLES);
    for (int offset = first; offset < points.Count; offset += SCENE_HIGHRES_BATCH_SAMPLES)
    {
        int end = Math.Min(points.Count, offset + SCENE_HIGHRES_BATCH_SAMPLES);
        StringBuilder entities = new StringBuilder("[");
        for (int i = offset; i < end; i++)
        {
            if (i > offset) entities.Append(',');
            entities.Append(points[i].Json);
        }
        entities.Append(']');
        QueueEvent("scene_highres_batch", "", ObjectJson(Field("trigger", Quote(trigger)), Field("entities", entities.ToString())));
    }
}

private void SampleStates(float now)
{
    IPlayer[] players = Game.GetPlayers();
    for (int i = 0; i < players.Length; i++)
    {
        IPlayer player = players[i];
        string sessionId = SessionId(player);
        if (string.IsNullOrEmpty(sessionId)) continue;
        string json = PlayerHighresJson(player);
        List<StatePoint> points;
        if (!m_ring.TryGetValue(sessionId, out points))
        {
            points = new List<StatePoint>();
            m_ring.Add(sessionId, points);
        }
        points.Add(new StatePoint { GameMs = now, Json = json });
        while (points.Count > 0 && now - points[0].GameMs > HIGHRES_WINDOW_MS) points.RemoveAt(0);
        float captureEnd;
        if (m_captureUntil.TryGetValue(sessionId, out captureEnd))
        {
            if (now <= captureEnd) QueueEvent("highres_state_sample", sessionId, json);
            else m_captureUntil.Remove(sessionId);
        }
    }
}

}

private void TriggerHighres(string sessionId, string trigger)
{
    if (string.IsNullOrEmpty(sessionId)) return;
    float now = Game.TotalElapsedGameTime;
    float last;
    if (m_lastHighresDump.TryGetValue(sessionId, out last) && now - last < 1000f)
    {
        m_captureUntil[sessionId] = Math.Max(m_captureUntil.ContainsKey(sessionId) ? m_captureUntil[sessionId] : 0, now + HIGHRES_POST_EVENT_MS);
        return;
    }
    m_lastHighresDump[sessionId] = now;
    m_captureUntil[sessionId] = now + HIGHRES_POST_EVENT_MS;
    List<StatePoint> points;
    if (!m_ring.TryGetValue(sessionId, out points)) return;
    StringBuilder samples = new StringBuilder("[");
    int first = Math.Max(0, points.Count - MAX_HIGHRES_WINDOW_SAMPLES);
    for (int i = first; i < points.Count; i++)
    {
        if (i > first) samples.Append(',');
        samples.Append(points[i].Json);
    }
    samples.Append(']');
    QueueEvent("highres_state_window", sessionId, ObjectJson(
        Field("trigger", Quote(trigger)),
        Field("window_before_ms", HIGHRES_WINDOW_MS.ToString()),
        Field("sample_limit", MAX_HIGHRES_WINDOW_SAMPLES.ToString()),
        Field("samples", samples.ToString())));
}

private PlayerSession EnsureSession(IUser user, string eventType)
{
    PlayerSession session = GetSession(user);
    if (session == null)
    {
        session = CreateSession(user);
        QueueEvent(eventType, session.Id, UserJson(user, session));
    }
    else
    {
        QueueEvent(eventType, session.Id, UserJson(user, session));
    }
    IPlayer player = user.GetPlayer();
    if (player != null) s_playerSessions[player.UniqueID] = session.Id;
    return session;
}

private PlayerSession CreateSession(IUser user)
{
    PlayerSession session = new PlayerSession
    {
        Id = Guid.NewGuid().ToString("D"),
        UserIdentifier = user.UserIdentifier,
        JoinedUtc = DateTime.UtcNow,
        AccountName = user.AccountName,
        CharacterName = user.Name
    };
    s_sessions[user.UserIdentifier] = session;
    return session;
}

private PlayerSession GetOrCreateSession(IUser user)
{
    PlayerSession session = GetSession(user);
    return session ?? CreateSession(user);
}

private PlayerSession GetSession(IUser user)
{
    if (user == null) return null;
    PlayerSession result;
    s_sessions.TryGetValue(user.UserIdentifier, out result);
    return result;
}

private string SessionId(IPlayer player)
{
    if (player == null) return "";
    string result;
    if (s_playerSessions.TryGetValue(player.UniqueID, out result)) return result;
    IUser user = player.GetUser();
    if (user == null) return "";
    PlayerSession session = GetOrCreateSession(user);
    s_playerSessions[player.UniqueID] = session.Id;
    return session.Id;
}

private void EndRound(string reason)
{
    if (m_roundEnded) return;
    m_roundEnded = true;
    IPlayer[] players = Game.GetPlayers();
    List<IPlayer> alive = new List<IPlayer>();
    for (int i = 0; i < players.Length; i++)
        if (!players[i].IsDead && !players[i].IsRemoved) alive.Add(players[i]);
    string resultSource = "unknown";
    bool draw = false;
    StringBuilder winners = new StringBuilder("[");
    if (Game.IsGameOver && alive.Count == 0)
    {
        draw = true;
        resultSource = "inferred_no_alive_players";
    }
    else if (Game.IsGameOver && alive.Count == 1 && alive[0].GetTeam() == PlayerTeam.Independent)
    {
        winners.Append(Quote(SessionId(alive[0])));
        resultSource = "inferred_alive_state";
    }
    else if (Game.IsGameOver && alive.Count > 0)
    {
        PlayerTeam team = alive[0].GetTeam();
        bool oneTeam = team != PlayerTeam.Independent;
        for (int i = 1; i < alive.Count; i++) if (alive[i].GetTeam() != team) oneTeam = false;
        if (oneTeam)
        {
            resultSource = "inferred_alive_state";
            for (int i = 0; i < alive.Count; i++)
            {
                if (i > 0) winners.Append(',');
                winners.Append(Quote(SessionId(alive[i])));
            }
        }
    }
    winners.Append(']');
    QueueEvent("round_end", "", ObjectJson(
        Field("reason", Quote(reason)),
        Field("duration_ms", Number(Game.TotalElapsedGameTime - m_roundStartMs)),
        Field("is_game_over", Bool(Game.IsGameOver)),
        Field("sudden_death_active", Bool(Game.SuddenDeathActive)),
        Field("draw", Bool(draw)),
        Field("winners", winners.ToString()),
        Field("result_source", Quote(resultSource))));
}

private void QueueHealth()
{
    QueueEvent("telemetry_health", "", ObjectJson(
        Field("dropped_event_count", m_droppedEventCount.ToString()),
        Field("dropped_low_priority_event_count", m_droppedLowPriorityEventCount.ToString()),
        Field("pending_event_count", m_pending.Count.ToString()),
        Field("pending_capacity", MAX_PENDING_EVENTS.ToString()),
        Field("flush_interval_ms", STORAGE_FLUSH_INTERVAL_MS.ToString()),
        Field("scene_dropped_sample_count", m_sceneDroppedSampleCount.ToString()),
        Field("scene_window_suppressed_count", m_sceneWindowSuppressedCount.ToString()),
        Field("scene_transport_tokens", Number(m_sceneTransportTokens)),
        Field("scene_bytes_last_minute", m_sceneBytesSinceHealth.ToString()),
        Field("storage_slot", m_storageSlot.ToString())));
    m_lastReportedDroppedCount = m_droppedEventCount;
    m_sceneBytesSinceHealth = 0;
}

private void QueueEvent(string type, string playerSessionId, string dataJson)
{
    long sequence = ++s_sequence;
    string envelope = ObjectJson(
        Field("v", "1"),
        Field("seq", sequence.ToString()),
        Field("ts", Quote(DateTime.UtcNow.ToString("o"))),
        Field("type", Quote(type)),
        Field("server_session", Quote(s_serverSessionId)),
        Field("round", Quote(m_roundId)),
        Field("player", string.IsNullOrEmpty(playerSessionId) ? "null" : Quote(playerSessionId)),
        Field("game_ms", Number(Game.TotalElapsedGameTime)),
        Field("round_ms", Number(Game.TotalElapsedGameTime - m_roundStartMs)),
        Field("data", dataJson));
    string raw = PREFIX + envelope;
    if (IsLowPrioritySceneEvent(type) && !TrySpendSceneTransport(raw.Length))
    {
        m_sceneDroppedSampleCount++;
        return;
    }
    if (m_pending.Count >= MAX_PENDING_EVENTS)
    {
        int lowPriorityIndex = FindOldestLowPriorityEvent();
        if (IsCriticalEvent(type) && lowPriorityIndex >= 0)
        {
            m_pending.RemoveAt(lowPriorityIndex);
            m_pendingTypes.RemoveAt(lowPriorityIndex);
            m_droppedLowPriorityEventCount++;
        }
        else
        {
            if (type == "scene_frame_batch" || type == "scene_window_complete") m_sceneDroppedSampleCount++;
            m_droppedEventCount++;
            return;
        }
    }
    if (IsSceneEvent(type)) m_sceneBytesSinceHealth += raw.Length;
    m_pending.Add(Convert.ToBase64String(Encoding.UTF8.GetBytes(raw)));
    m_pendingTypes.Add(type);
    if (VERBOSE_DEBUG && type != "state_sample" && type != "highres_state_sample")
        Game.WriteToConsole("SFDTelemetry " + type + " #" + sequence.ToString());
}

private bool IsLowPrioritySceneEvent(string type)
{
    return type == "scene_frame_batch";
}

private bool IsSceneEvent(string type)
{
    return type == "scene_manifest_batch" || type == "scene_frame_batch" || type == "scene_highres_batch" || type == "scene_window_complete" ||
        type == "object_created" || type == "object_damage" || type == "object_damage_batch" || type == "object_terminated" ||
        type == "projectile_created" || type == "projectile_hit" || type == "explosion_hit" || type == "melee_action";
}

private bool TrySpendSceneTransport(int bytes)
{
    float now = Game.TotalElapsedGameTime;
    float elapsed = Math.Max(0f, now - m_sceneTransportLastMs);
    m_sceneTransportLastMs = now;
    m_sceneTransportTokens = Math.Min(SCENE_TRANSPORT_BURST_BYTES, m_sceneTransportTokens + elapsed / 1000f * SCENE_TRANSPORT_AVG_BYTES_PER_SECOND);
    if (bytes > m_sceneTransportTokens) return false;
    m_sceneTransportTokens -= bytes;
    return true;
}

private void FlushPending()
{
    if (m_pending.Count == 0 || m_transport == null) return;
    int count = Math.Min(MAX_EVENTS_PER_SLOT, m_pending.Count);
    string[] values = m_pending.GetRange(0, count).ToArray();
    m_pending.RemoveRange(0, count);
    m_pendingTypes.RemoveRange(0, count);
    m_transport.SetItem("slot_" + m_storageSlot.ToString(), values);
    m_storageSlot = (m_storageSlot + 1) % STORAGE_SLOT_COUNT;
    if (m_droppedEventCount != m_lastReportedDroppedCount) QueueHealth();
}

private bool IsCriticalEvent(string type)
{
    return type == "player_damage" || type == "player_death" || type == "round_start" ||
        type == "round_end" || type == "user_join" || type == "user_leave" ||
        type == "chat_message" || type == "object_created" || type == "object_damage" || type == "object_damage_batch" ||
        type == "object_terminated" || type == "projectile_created" || type == "projectile_hit" ||
        type == "explosion_hit" || type == "melee_action" || type == "scene_window_complete" || type == "scene_highres_batch";
}

private int FindOldestLowPriorityEvent()
{
    for (int i = 0; i < m_pendingTypes.Count; i++)
    {
        string type = m_pendingTypes[i];
        if (type == "key_input_batch" || type == "key_input" || type == "state_sample" || type == "scene_frame_batch" || type == "scene_window_complete" ||
            type == "highres_state_sample" || type == "highres_state_window") return i;
    }
    return -1;
}

private void ForceFlush()
{
    int safety = 0;
    while (m_pending.Count > 0 && safety < (MAX_PENDING_EVENTS / MAX_EVENTS_PER_SLOT + 2))
    {
        FlushPending();
        safety++;
    }
}

private string Field(string name, string rawValue)
{
    return Quote(name) + ":" + (rawValue ?? "null");
}

private string ObjectJson(params string[] fields)
{
    StringBuilder result = new StringBuilder("{");
    for (int i = 0; i < fields.Length; i++)
    {
        if (i > 0) result.Append(',');
        result.Append(fields[i]);
    }
    result.Append('}');
    return result.ToString();
}

private string Quote(string value)
{
    if (value == null) return "null";
    StringBuilder result = new StringBuilder(value.Length + 2);
    result.Append('"');
    for (int i = 0; i < value.Length; i++)
    {
        char c = value[i];
        if (c == '"') result.Append("\\\"");
        else if (c == '\\') result.Append("\\\\");
        else if (c == '\b') result.Append("\\b");
        else if (c == '\f') result.Append("\\f");
        else if (c == '\n') result.Append("\\n");
        else if (c == '\r') result.Append("\\r");
        else if (c == '\t') result.Append("\\t");
        else if (c < 32) result.Append("\\u" + ((int)c).ToString("x4"));
        else result.Append(c);
    }
    result.Append('"');
    return result.ToString();
}

private string Bool(bool value) { return value ? "true" : "false"; }
private string Number(float value) { return value.ToString("R", System.Globalization.CultureInfo.InvariantCulture); }
private string Number(double value) { return value.ToString("R", System.Globalization.CultureInfo.InvariantCulture); }
